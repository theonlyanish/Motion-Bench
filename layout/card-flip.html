Card Flip

3D rotation on click/hover

HTML · CSS · No dependencies · MIT

HTML

<div class="flip-card">
  <div class="flip-inner">
    <div class="flip-front">Front</div>
    <div class="flip-back">Back</div>
  </div>
</div>

CSS

.flip-card {
  width: 140px;
  height: 180px;
  perspective: 600px; /* gives the rotation its 3D depth */
  cursor: pointer;
  color: #f0f0f5;
}

.flip-inner {
  width: 100%;
  height: 100%;
  position: relative;
  transform-style: preserve-3d;
  transition: transform 0.6s cubic-bezier(0.16, 1, 0.3, 1);
}

.flip-card:hover .flip-inner {
  transform: rotateY(180deg);
}

.flip-front,
.flip-back {
  position: absolute;
  width: 100%;
  height: 100%;
  backface-visibility: hidden; /* hides the reversed side */
  display: flex;
  align-items: center;
  justify-content: center;
  border-radius: 8px;
  border: 1px solid rgba(255, 255, 255, 0.06);
}

.flip-front {
  background: #222;
}

.flip-back {
  background: #7c5cff;
  transform: rotateY(180deg); /* pre-flipped so it reads correctly */
  color: #fff;
}
