Staggered Reveal

Letters animate in with stagger — like Framer's staggerChildren

HTML · CSS · JavaScript · No dependencies · MIT

HTML

<p class="stagger-headline" id="staggerHeadline"></p>

CSS

.stagger-headline {
  display: flex;
  justify-content: center;
  font-size: 3.5rem;
  font-weight: 600;
}

.stagger-char {
  display: inline-block;
  opacity: 0;
  transform: translateY(24px);
  will-change: transform, opacity;
}

.stagger-char.revealed {
  opacity: 1;
  transform: translateY(0);
}

JavaScript

var headline = document.getElementById('staggerHeadline');
var text = 'Hello World';
var DELAY = 50; // ms between letters
var EASE = 'cubic-bezier(0.16, 1, 0.3, 1)';

headline.innerHTML = text.split('').map(function (c) {
  return '<span class="stagger-char">' + (c === ' ' ? '&nbsp;' : c) + '</span>';
}).join('');
var chars = headline.querySelectorAll('.stagger-char');

requestAnimationFrame(function () {
  chars.forEach(function (el, i) {
    el.style.transition =
      'opacity 0.5s ' + EASE + ' ' + (i * DELAY) + 'ms, ' +
      'transform 0.5s ' + EASE + ' ' + (i * DELAY) + 'ms';
    el.classList.add('revealed');
  });
});
